       <div class="pname"><span>${p.name}</span><span class="tag ${p.statusCls}">${p.status}</span></div>
       <div class="ptype">${p.type}</div>
       <div class="prev"><span>REVISION ${p.rev}</span><span class="pid">${p.id}</span></div>
     </div>`).join('');
  document.querySelectorAll('.proj-item').forEach(el =>
    el.addEventListener('click', () => switchProject(el.dataset.id)));
}

function renderSelect(){
  $('project-select').innerHTML = PROJECTS.map(p =>
    `<option value="${p.id}" ${p.id===currentId?'selected':''}>${p.name}</option>`).join('');
}

function renderCandidates(){
  const p = proj();
  const strip = $('cand-strip');
  if (!p.candidates.length){ strip.innerHTML = '<div class="empty-note" style="padding:10px 12px">尚无候选方向</div>'; return; }
  strip.innerHTML = p.candidates.map(c =>
    `<div class="cand-card ${c.id===selectedCand?'selected':''} ${c.status==='已退回'?'rejected':''}" data-id="${c.id}">
       <div class="ct">${c.title}</div>
       <div class="cd">${c.desc}</div>
       <div class="cf">
         <span class="st ${c.status==='待审'?'st-pending':'st-rejected'}">${c.status}</span>
         ${c.tweaks ? `<span class="tw">微调 ×${c.tweaks}</span>` : '<span></span>'}
       </div>
     </div>`).join('');
  document.querySelectorAll('.cand-card').forEach(el =>
    el.addEventListener('click', () => { selectedCand = el.dataset.id; renderCandidates(); renderPreview(); }));
}

function renderPreview(){
  const p = proj();
  const canvas = $('canvas');
  const c = p.candidates.find(x => x.id === selectedCand) || p.candidates.find(x => x.status === '待审') || p.candidates[0];
  if (!c){
    canvas.innerHTML = `<div class="canvas-empty">
        <div class="ce-title">暂无可预览的候选</div>
        <div class="ce-sub">该项目还没有生成候选方向，或正在等待你的澄清回复。</div>
      </div>`;
    return;
  }
  canvas.innerHTML = `<div class="canvas-label">${c.title} · ${c.status}</div>` +
    c.blocks.map(([x,y,w,h,col]) =>
      `<div class="wf-block" style="left:${x}%;top:${y}%;width:${w}%;height:${h}%;background:${col}"></div>`).join('');
}

function renderTimeline(){
  const p = proj();
  $('tl-count').textContent = p.timeline.length + ' 条';
  $('timeline').innerHTML = p.timeline.map(([t,r,m]) =>
    `<div class="tl-row"><span class="tl-time">${t}</span><span class="tl-role ${r}">${r}</span><span class="tl-msg">${m}</span></div>`).join('');
}

function renderConstraints(){
  const c = proj().constraints;
  $('constraints').innerHTML =
    `<div class="kv"><span class="k">约束预设</span><span class="v">${c.preset}</span></div>
     <div class="kv"><span class="k">模板角色</span><span class="v">${c.role}</span></div>
     <ul class="rules">${c.rules.map(r=>`<li>${r}</li>`).join('')}</ul>`;
}

function renderQuality(){
  const q = proj().quality;
  const track = (cls, name, sub, items) =>
    `<div class="track ${cls}">
       <div class="track-head"><span class="track-name">${name}</span><span class="track-count">${items.length}</span></div>
       <div class="track-sub">${sub}</div>
       ${items.length ? items.map(i=>`<div class="track-item">${i}</div>`).join('') : '<div class="track-none">无</div>'}
     </div>`;
  $('quality').innerHTML =
    track('track-hard','硬错误','客观问题 · 未清零前锁定导出', q.hard) +
    track('track-aes','审美发现','主观建议 · 不挡交付，由你判断', q.aes);
  const ok = q.hard.length === 0;
  $('export-lock').innerHTML = q.exported ? '导出：<b>已交付</b>' : (ok ? '导出：<b>已解锁</b>' : '导出：未进入复核 / 硬错误未清零');
  $('export-btn').disabled = !(ok || q.exported);
}

function renderDeliveries(){
  const d = proj().deliveries;
  $('dl-count').textContent = d.length;
  $('deliveries').innerHTML = d.length
    ? d.map(([t,m]) => `<div class="dl-item"><b>${t}</b> · ${m}</div>`).join('')
    : '<div class="empty-note">尚无交付记录</div>';
}

function renderFeedbackTargets(){
  const p = proj();
  $('fb-b-target').innerHTML = p.candidates.length
    ? p.candidates.map(c=>`<option>${c.title}</option>`).join('')
    : '<option>（暂无候选）</option>';
}

function renderAll(){
  renderHeader(); renderProjects(); renderSelect(); renderCandidates();
  renderPreview(); renderTimeline(); renderConstraints(); renderQuality();
  renderDeliveries(); renderFeedbackTargets();
}

/* ================= 交互 ================= */
function switchProject(id){
  if (id === currentId) return;
  currentId = id;
  const p = proj();
  selectedCand = (p.candidates.find(c=>c.status==='待审') || p.candidates[0] || {}).id || null;
  document.body.style.opacity = '.45';
  setTimeout(() => { document.body.style.opacity = '1'; renderAll(); }, 200 + Math.random()*300);
}
$('project-select').addEventListener('change', e => switchProject(e.target.value));

function pushEvent(role, msg){
  proj().timeline.unshift([nowTime().slice(0,5), role, msg]);
  renderTimeline();
}
$('fb-a-submit').addEventListener('click', () => {
  const dim = $('fb-a-dim').value, note = $('fb-a-note').value.trim();
  pushEvent('USER', `退回整体方向（${dim}）${note ? '：' + note : ''}`);
  $('fb-a-note').value = '';
});
$('fb-b-submit').addEventListener('click', () => {
  const t = $('fb-b-target').value, pos = $('fb-b-pos').value.trim(), note = $('fb-b-note').value.trim();
  if (!pos && !note) return;
  pushEvent('USER', `局部微调：${t}${pos ? ' · ' + pos : ''}${note ? ' — ' + note : ''}`);
  $('fb-b-pos').value = ''; $('fb-b-note').value = '';
});
$('export-btn').addEventListener('click', () => {
  if ($('export-btn').disabled) return;
  pushEvent('SYS', '导出交付：Artifact 已打包导出');
});

renderAll();
