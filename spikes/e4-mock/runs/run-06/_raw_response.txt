      if (DB.projects[i].id === id) return DB.projects[i];
    }
    return null;
  }
  function excerpt(text, n) {
    text = String(text).replace(/\s+/g, ' ').trim();
    return text.length > n ? text.slice(0, n) + '…' : text;

   * Agent 回复生成器（规则模板推导，非写死文案）

   * 规则：
   *   1. 消息含「组件/代码/按钮/函数/code」→ 生成 code 类产出 + 代码说明回复；

   *   3. 含「计划/步骤/清单/方案/安排/拆解」→ 生成 steps 类任务清单 + 说明回复；
   *   4. 其他                               → 生成 card 类"理解摘要"卡片 + 摘要回复。
   * 所有模板都会把用户原文片段、当前项目名、消息序号注入，因此同样一句话
   * 在不同项目里、或同一项目里说多次，得到的内容也不完全一样。
   * ========================================================================= */
  function generateReply(project, userText) {
    var turn = project.messages.length + 1;
    var quote = excerpt(userText, 18);

    if (/组件|代码|按钮|函数|脚本|code/i.test(userText)) {
      return {
        text: '明白，我把「' + quote + '」当作一个编码任务来处理。结合「' + project.name + '」的上下文，我生成了一段可直接使用的实现（第 ' + turn + ' 轮产出），代码已同步到右侧预览区；如果你想换语言或调整参数命名，继续说就行。',
        artifact: {

            '// 由 ' + project.name + ' · 第 ' + turn + ' 轮生成\n' +
            'function buildSolution() {\n' +
            '  const input = "' + excerpt(userText, 24).replace(/"/g, '\\"') + '";\n' +
            '  const steps = input.split(/[，,。.;；]/).filter(Boolean);\n' +
            '  return steps.map((s, i) => ({ order: i + 1, task: s.trim(), done: false }));\n' +
            '}\n\n' +
            'module.exports = { buildSolution };'
        }
      };
    }

    if (/报告|周报|日报|数据|统计|指标|分析/.test(userText)) {
      var seed = (userText.length * 7 + turn * 13) % 90;
      return {
        text: '好的，我围绕「' + quote + '」整理了一份数据视角的摘要卡：核心数值基于该会话第 ' + turn + ' 轮的上下文推算（演示环境为规则生成），风险项和建议已一并写入，完整卡片见预览区。',
        artifact: {
          kind: 'card',
          title: '数据摘要 · ' + project.name,
          desc: '主题：' + excerpt(userText, 40) + '\n\n整体表现平稳，综合得分 ' + (60 + seed % 35) + '/100。发现 1 个需要关注的波动项，建议先核对口径再下结论；下轮可补充对比维度做交叉验证。',
          stats: [

            { num: String(seed % 4 + 1), label: '风险项' },

          ]
        }
      };
    }

    if (/计划|步骤|清单|方案|安排|拆解|流程/.test(userText)) {
      return {
        text: '我把「' + quote + '」拆成了一份 4 步执行清单（这是该会话第 ' + turn + ' 次拆解，粒度比上一轮更细）。每一步都带了验收标准，右侧预览区可以逐项查看，觉得节奏太紧或太松都可以让我重排。',

      artifact: {
        kind: 'card',
        title: '理解摘要 · 第 ' + turn + ' 轮',
        desc: '原始输入：' + excerpt(userText, 60) + '\n\n核心诉求：围绕该主题获得可执行的下一步。待确认：① 期望的产出形态（文档 / 代码 / 清单）；② 时间或范围约束。',

# MOCK.md —— Mock 行为声明

本原型（NovaDesk · Agent 工作台）**没有任何真实后端**。所有数据与"网络请求"均由前端
`mock-api.js` 文件内伪造的 Mock 层提供。本文件如实、完整声明代码中实际存在的全部

## 1. 被 Mock 的接口 / 交互点

Mock 层以 `window.__mockApi__` 暴露，模拟了 4 个 REST 风格端点，覆盖界面上全部三块交互：

| 暴露方法 | 模拟的端点 | 用途 | 界面上的表现 |
|---|---|---|---|
| `listProjects()` | `GET /api/projects` | 获取项目列表摘要 | 左侧项目列表的首次加载 |
| `getProject(id)` | `GET /api/projects/:id` | 获取单个项目完整数据（含消息历史） | 点击列表项切换「当前项目」，对话区与预览区随之刷新 |
| `sendMessage(projectId, text)` | `POST /api/projects/:id/messages` | 发送一条用户消息并返回 agent 回复 | 对话区追加用户气泡 + agent 回复气泡；预览区更新产出 |
| `createProject(name)` | `POST /api/projects` | 新建项目 | 「＋ 新建项目」按钮，新项目插入列表顶部并自动选中 |

界面层（`app.js`）只调用 `window.__mockApi__`，不直接读写数据，模拟了前后端分离的调用方式。

## 2. 数据来源声明

### 2.1 纯虚构编造的数据

- **4 个预置项目**（官网落地页生成 / 每周数据周报 / React 组件工坊 / 发布计划编排）的
  名称、描述、状态、更新时间：全部为手写虚构，不对应任何真实产品或数据。
- **每个项目的预置历史消息**（用户提问 + agent 回答文案）：全部为手写虚构文案。
- **每条预置 agent 消息附带的产出 artifact**（落地页大纲卡、数据周报卡、
  AsyncButton.jsx 代码、灰度上线 5 步清单）：全部为手写虚构内容，其中周报里的
  DAU / 留存等数字是随意编造的演示数字，不来自任何真实统计。

这些数据以**内联数组**（`mock-api.js` 中的 `DB` 常量）形式硬编码在代码里。

### 2.2 按规则/模板推导生成的数据

`sendMessage()` 返回的 agent 回复**不是写死的**，由 `generateReply()` 按以下规则从模板推导：

1. 用户消息含关键词 `组件/代码/按钮/函数/脚本/code` → 生成 **code 类产出**：
   代码内容是把用户原文按标点切分后映射为任务数组的 JS 片段，文件名由
   项目 id + 会话轮次拼接。
2. 含 `报告/周报/日报/数据/统计/指标/分析` → 生成 **card 类数据卡片**：
   卡片中的数值由 `用户输入长度 × 7 + 轮次 × 13` 对 90 取模得到的伪随机种子推导，
   因此每次结果不同但确定性可复现。
3. 含 `计划/步骤/清单/方案/安排/拆解/流程` → 生成 **steps 类任务清单**：
   固定 4 步模板，其中第 1、4 步会把用户原文片段和项目名注入。
4. 其他消息 → 生成 **card 类"理解摘要"卡片**。

所有模板都会注入：用户原文摘要片段、当前项目名、当前会话轮次序号，
所以同一句话在不同项目或不同轮次下得到的回复内容不同。
`createProject()` 生成的欢迎消息与初始卡片也由项目名参数化模板生成。

### 2.3 界面派生数据（非 mock 接口，属正常前端逻辑）

- 列表项的"最近一条消息"摘要：从 `listProjects()` 返回数据的截断，属正常展示逻辑。
- 预览区内容：取当前项目消息历史中**最近一条带 artifact 的 agent 消息**渲染，
  因此切换项目、或发送消息产生新回复后，预览都会真实变化。

## 3. 技术手段声明

代码中实际使用的 mock 技术手段，共 3 种，无遗漏：

1. **内联假数据数组**：`mock-api.js` 顶部的 `DB` 常量，作为"数据库"。
2. **`Promise` + `setTimeout` 模拟异步网络延迟**：`respond()` 包装器给每次调用加
   300~900ms 的随机延迟（`latency()`），配合界面上的"Agent 思考中…"状态与打字动画，
   模拟真实请求耗时。**没有**使用 fetch 拦截、XMLHttpRequest 重写或 Service Worker。
3. **`window.__mockApi__` 假客户端**：以全局对象形式暴露上述 4 个方法，
   界面层像调用真实 SDK 一样调用它，后续若接真实后端只需替换该对象的实现。

## 4. 已知边界（如实声明）

- 数据仅存在于页面内存中，**刷新页面后**发送的消息和新建的项目会丢失（没有使用
  localStorage 等持久化手段）。
- "乐观渲染"的用户消息时间先显示为"刚刚"，待 mock 回包后修正为真实时分。
- 列表搜索仅为前端过滤，不走 mock 接口。
