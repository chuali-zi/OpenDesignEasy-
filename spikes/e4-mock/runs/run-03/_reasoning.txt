fetch override:

```js
const nativeFetch = window.fetch ? window.fetch.bind(window) : null;
window.fetch = function(input, init){
  const url = typeof input === 'string' ? input : input.url;
  const method = (init && init.method) || (typeof input!=='string' && input.method) || 'GET';
  const body = init && init.body;
  if (url.startsWith('/api/') || url.includes('://mock.local/api/')) {
    return routeRequest(method.toUpperCase(), url, body).then(
      payload => jsonResponse(payload, 200),
      err => jsonResponse({error: err.message}, err.status||500)
    );
  }
  if (nativeFetch) return nativeFetch(input, init);
  return Promise.reject(new Error('No network in mock environment'));
};
```

jsonResponse uses `new Response(JSON.stringify(data), {status, headers:{'Content-Type':'application/json','X-Mock':'orbit'}})`.

Router:

```js
function routeRequest(method, url, body){
  const path = url.replace(/^https?:\/\/[^/]+/,'');
  let m;
  if (method==='GET' && path==='/api/projects') return ok(listProjects());
  if ((m=path.match(/^\/api\/projects\/([\w-]+)$/)) && method==='GET') return ok(getProject(m[1]));
  if ((m=path.match(/^\/api\/projects\/([\w-]+)\/messages$/))){
    if(method==='GET') return ok(getMessages(m[1]));
    if(method==='POST') return ok(postMessage(m[1], JSON.parse(body)));
  }
  if ((m=path.match(/^\/api\/projects\/([\w-]+)\/preview$/)) && method==='GET') return ok(getPreview(m[1]));
  return Promise.reject(Object.assign(new Error('404 mock: '+path),{status:404}));
}
```

function generateAgentReply(project, userText, history){
