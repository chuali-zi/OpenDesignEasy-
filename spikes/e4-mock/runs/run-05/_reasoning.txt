```js
function craftReply(project, text) {
  var rule = pickRule(text);
  var pv = db.previews[project.id];
  var summary = text.replace(/\s+/g, ' ');
  if (summary.length > 20) summary = summary.slice(0, 20) + '…';
  var n = pv.revisions.length + 1;
  pv.version += 1;
  var comment = commentFor(pv.language, 'rev ' + n + ': ' + summary);
  pv.code = pv.code + '\n' + comment;
  pv.revisions.push({ n: n, summary: summary, ts: Date.now() });
  var quote = text.replace(/\s+/g,' ');
  if (quote.length > 16) quote = quote.slice(0,16)+'…';
  var body = '收到。我把「' + quote + '」归类为「' + rule.tag + '」，将在「' + project.name + '」上按以下顺序执行：\n\n' +
    rule.steps.map(function(s,i){return (i+1)+'. '+s;}).join('\n') +
    '\n\n改动已落到预览产物：修订 #' + n + '，版本升至 v0.' + pv.version + '。你可以继续补充约束，我会在现有结果上增量调整。';
  return { id: uid('m'), role: 'agent', text: body, ts: Date.now() };
}

function commentFor(lang, s) {
  if (lang === 'javascript') return '// ' + s;
  return '<!-- ' + s + ' -->';
}
```
